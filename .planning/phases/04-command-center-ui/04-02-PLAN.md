```ts
import { create } from 'zustand'

export type Theme = 'dark' | 'light'
export type Language = 'en' | 'tr'
export type WsStatus = 'idle' | 'connecting' | 'connected' | 'reconnecting' | 'disconnected'

interface UIState {
  theme: Theme
  language: Language
  sidebarCollapsed: boolean
  wsStatus: WsStatus
  setTheme: (theme: Theme) => void
  toggleTheme: () => void
  setLanguage: (lang: Language) => void
  toggleSidebar: () => void
  setWsStatus: (status: WsStatus) => void
}

// D-02: dark default. Language detected from navigator during i18n init.
export const useUIStore = create<UIState>((set) => ({
  theme: 'dark',
  language: 'en',
  sidebarCollapsed: false,
  wsStatus: 'idle',
  setTheme: (theme) => set({ theme }),
  toggleTheme: () => set((s) => ({ theme: s.theme === 'dark' ? 'light' : 'dark' })),
  setLanguage: (language) => set({ language }),
  toggleSidebar: () => set((s) => ({ sidebarCollapsed: !s.sidebarCollapsed })),
  setWsStatus: (wsStatus) => set({ wsStatus }),
}))
```

Write `web/src/hooks/useTheme.ts`:
```ts
import { useEffect } from 'react'
import { useUIStore } from '@/stores/ui-store'

// Mounts once in ThemeProvider. Syncs .dark class on <html> with store.theme.
export function useTheme() {
  const theme = useUIStore((s) => s.theme)
  useEffect(() => {
    const root = document.documentElement
    root.classList.toggle('dark', theme === 'dark')
  }, [theme])
}
```

```tsx
                      to={item.to}
                      className={cn(

                          : 'text-zinc-400 hover:bg-zinc-800 hover:text-zinc-50',
                      )}
                    >
                      <Icon className={cn('size-5', active && 'text-emerald-500')} />
                      {!collapsed && <span>{t(item.key)}</span>}
                    </Link>
                  </li>
                )
              })}
            </ul>
          </div>
        ))}
      </nav>

  )
}
```
