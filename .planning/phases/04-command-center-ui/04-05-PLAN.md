import { cn } from '@/lib/utils'
import type { AgentStatus, TaskStatus } from '@/types/entities'

// UI-SPEC status color map (verbatim tokens).
const agentTokens: Record<AgentStatus, { dot: string; bg: string; text: string }> = {
  online: { dot: 'bg-emerald-500', bg: 'bg-emerald-500/10', text: 'text-emerald-400' },
  idle: { dot: 'bg-amber-500', bg: 'bg-amber-500/10', text: 'text-amber-400' },
  offline: { dot: 'bg-zinc-500', bg: 'bg-zinc-500/10', text: 'text-zinc-400' },
  error: { dot: 'bg-red-500', bg: 'bg-red-500/10', text: 'text-red-400' },
}

const taskTokens: Record<TaskStatus, { dot: string; text: string; pulse?: boolean; strike?: boolean }> = {
  queued: { dot: 'bg-zinc-400', text: 'text-zinc-400' },
  claimed: { dot: 'bg-violet-400', text: 'text-violet-400' },
  running: { dot: 'bg-sky-400', text: 'text-sky-400', pulse: true },
  completed: { dot: 'bg-emerald-500', text: 'text-emerald-400' },
  failed: { dot: 'bg-red-500', text: 'text-red-400' },
  cancelled: { dot: 'bg-zinc-500', text: 'text-zinc-500', strike: true },
}

export function AgentStatusBadge({ status, label }: { status: AgentStatus; label: string }) {
  const tok = agentTokens[status]
  return (
    <span className={cn('inline-flex items-center gap-1.5 rounded-md px-2 py-0.5 text-xs font-medium', tok.bg, tok.text)}>
      <span className={cn('size-1.5 rounded-full', tok.dot)} />
      {label}
    </span>
  )
}

export function TaskStatusBadge({ status, label }: { status: TaskStatus; label: string }) {
  const tok = taskTokens[status]
  return (
    <span className={cn('inline-flex items-center gap-1.5 text-xs font-medium', tok.text, tok.strike && 'line-through')}>
      <span className={cn('size-1.5 rounded-full', tok.dot, tok.pulse && 'animate-pulse')} />
      {label}
    </span>
  )
}

Write `web/src/components/common/ResponsiveList.tsx`:
```tsx
import { ReactNode } from 'react'

interface Column<T> {
  key: keyof T | string
  label: string
  render?: (item: T) => ReactNode
}

interface Props<T extends { id: string }> {
  items: T[]
  columns: Column<T>[]
  cardRender: (item: T) => ReactNode
  emptyState?: ReactNode
}

// UI-15: table on md+ (hidden md:block), cards on <md (md:hidden).
// UI-SPEC Data table density: py-2 px-3, text-zinc-400 header, hover:bg-zinc-800/50.
export function ResponsiveList<T extends { id: string }>({ items, columns, cardRender, emptyState }: Props<T>) {
  if (items.length === 0 && emptyState) return <>{emptyState}</>
  return (
    <>
      <div className="hidden md:block">
        <table className="w-full border-collapse">
          <thead>
            <tr className="border-b border-zinc-800">
              {columns.map((c) => (
                <th
                  key={String(c.key)}
                  className="px-3 py-2 text-left text-xs font-medium uppercase tracking-wide text-zinc-400"
                >
                  {c.label}
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {items.map((i) => (
              <tr key={i.id} className="border-b border-zinc-800 hover:bg-zinc-800/50">
                {columns.map((c) => (
                  <td key={String(c.key)} className="px-3 py-2 text-sm text-zinc-50 tabular-nums">
                    {c.render ? c.render(i) : String((i as any)[c.key] ?? '')}
                  </td>
                ))}
              </tr>
            ))}
          </tbody>
        </table>
      </div>
      <div className="md:hidden space-y-2 px-4 py-2">
        {items.map((i) => (
          <div key={i.id} className="rounded-lg border border-zinc-800 bg-zinc-900 p-3">
            {cardRender(i)}
          </div>
        ))}
      </div>
    </>
  )
}
```

import { createFileRoute } from '@tanstack/react-router'
import { useWorkflow } from '@/hooks/queries/useWorkflows'
import { Badge } from '@/components/ui/badge'

export const Route = createFileRoute('/_authed/workflows/$workflowId')({
  component: WorkflowDetail,
})

function WorkflowDetail() {
  const { workflowId } = Route.useParams()
  const { data: workflow, isLoading } = useWorkflow(workflowId)

  if (isLoading) return <div className="p-6 text-sm text-zinc-400">Loading...</div>
  if (!workflow) return <div className="p-6 text-sm text-zinc-400">Workflow not found</div>

  return (
    <div className="p-6">
      <h1 className="mb-4 text-2xl font-semibold text-zinc-50">{workflow.name}</h1>
      <ol className="flex flex-col gap-2">
        {workflow.steps.map((step, idx) => (
          <li key={step.id} className="flex items-center gap-3 rounded-md border border-zinc-800 bg-zinc-900 p-3">
            <span className="font-mono text-xs text-zinc-400">{idx + 1}</span>
            <span className="flex-1 text-sm text-zinc-50">{step.name}</span>
            <Badge variant="outline">{step.status}</Badge>
          </li>
        ))}
      </ol>
    </div>
  )
}
