      ```python
      """Pydantic models for vector search API."""
      from typing import Any, Dict, List, Optional
      from pydantic import BaseModel, ConfigDict, Field

      ENTITY_TYPES = ["memory", "task", "artifact", "message"]


      class SearchRequest(BaseModel):
          model_config = ConfigDict(extra="forbid")
          query: str = Field(min_length=1, max_length=5000)
          types: Optional[List[str]] = Field(default=None, description="Entity types to search; default all")
          filters: Optional[Dict[str, Any]] = Field(default_factory=dict)
          top_k: int = Field(default=10, ge=1, le=50)


      class SearchHit(BaseModel):
          model_config = ConfigDict(extra="forbid")
          entity_type: str
          id: str
          content: str
          distance: float


      class SearchResponse(BaseModel):
          model_config = ConfigDict(extra="forbid")
          query: str
          total: int
          hits: List[SearchHit]
      ```

    Step B - Create `app/api/routes_search.py`:
      ```python
      """Unified vector search API (experimental - VEC-05)."""
      from typing import List
      import structlog
      from fastapi import APIRouter, Depends, HTTPException

      from ..database.connection import get_database
      from ..database.vector_availability import require_vector
      from ..models.vector_search import SearchRequest, SearchResponse, SearchHit, ENTITY_TYPES
      from ..services.embedding_service import get_embedding_service
      from ..services.vector_search_service import VectorSearchService

      logger = structlog.get_logger(__name__)

      router = APIRouter(
          prefix="/v1/search",
          tags=["search [experimental]"],
          dependencies=[Depends(require_vector)],
      )


      async def unified_search(req: SearchRequest) -> SearchResponse:
          backend = get_embedding_service()
          if backend is None:
              raise HTTPException(
                  status_code=503,
                  detail={
                      "type": "https://openhub.dev/problems/embedding-unavailable",
                      "title": "Embedding backend unavailable",
                      "status": 503,
                      "detail": "No embedding backend is configured. Check AGENTHUB_EMBEDDING_PROVIDER and AGENTHUB_OPENAI_API_KEY.",
                  },
              )

          query_vectors = await backend.embed([req.query])
          if not query_vectors:
              raise HTTPException(status_code=500, detail="Embedding backend returned empty result")
          qvec = query_vectors[0]

          types = req.types if req.types else ENTITY_TYPES[:]
          invalid = [t for t in types if t not in ENTITY_TYPES]
          if invalid:
              raise HTTPException(
                  status_code=400,
                  detail={
                      "type": "https://openhub.dev/problems/invalid-entity-type",
                      "title": "Invalid entity types",
                      "status": 400,
                      "detail": f"Unknown entity types: {invalid}. Valid: {ENTITY_TYPES}",
                  },
              )

          svc = VectorSearchService(get_database())
          all_hits: List[dict] = []
          for t in types:
              try:
                  hits = svc.search_entity(t, qvec, top_k=req.top_k, filters=req.filters or {})
                  all_hits.extend(hits)
              except Exception as e:
                  logger.warning("unified_search_entity_failed", entity_type=t, error=str(e)[:500])

          all_hits.sort(key=lambda h: h.get("distance", float("inf")))
          capped = all_hits[: req.top_k]
          return SearchResponse(
              query=req.query,
              total=len(capped),
              hits=[SearchHit(**h) for h in capped],
          )


      @router.post("", response_model=SearchResponse, summary="Unified semantic search (experimental)")
      async def post_search(req: SearchRequest) -> SearchResponse:
          return await unified_search(req)
      ```

    Step C - Extend `app/api/routes_search.py` with two new handlers on the SAME router:
      ```python
      from fastapi import Depends, HTTPException, Path
      from ..auth.dependencies import CurrentAdmin
      from ..models.vector_search import (
          ReindexRequest, ReindexResponse, ReindexByType,
          DeleteEmbeddingResponse, ENTITY_TYPES,
      )

      @router.post(
          "/reindex",
          response_model=ReindexResponse,
          summary="Re-embed unindexed entities (admin-only, experimental)",
      )
      async def reindex_embeddings(
          req: ReindexRequest,
          admin: CurrentAdmin,
      ) -> ReindexResponse:
          # Validate entity_type scope
          if req.entity_type is not None and req.entity_type not in ENTITY_TYPES:
              raise HTTPException(
                  status_code=400,
                  detail={
                      "type": "https://openhub.dev/problems/invalid-entity-type",
                      "title": "Invalid entity type",
                      "status": 400,
                      "detail": f"Unknown entity_type '{req.entity_type}'. Valid: {ENTITY_TYPES}",
                  },
              )
          backend = get_embedding_service()
          if backend is None:
              raise HTTPException(
                  status_code=503,
                  detail={
                      "type": "https://openhub.dev/problems/embedding-unavailable",
                      "title": "Embedding backend unavailable",
                      "status": 503,
                      "detail": "No embedding backend is configured; cannot re-index.",
                  },
              )
          svc = VectorSearchService(get_database())
          types = [req.entity_type] if req.entity_type else ENTITY_TYPES[:]
          reindexed = failed = skipped = 0
          by_type = {t: 0 for t in ENTITY_TYPES}
          for t in types:
              rows = svc.list_unindexed(entity_type=t, since=req.since)
              if not rows:
                  continue
              contents = [r["content"] for r in rows]
              try:
                  vectors = await backend.embed(contents)
              except Exception as e:
                  logger.error("reindex_batch_embed_failed", entity_type=t, error=str(e)[:500])
                  failed += len(rows)
                  continue
              for row, vec in zip(rows, vectors):
                  try:
                      svc.write_embedding(t, row["id"], vec, status="indexed")
                      reindexed += 1
                      by_type[t] += 1
                  except Exception as e:
                      logger.warning("reindex_write_failed", entity_type=t, id=row["id"], error=str(e)[:500])
                      failed += 1
          logger.info(
              "reindex_complete",
              admin_id=admin.agent_id if hasattr(admin, "agent_id") else None,
              reindexed=reindexed, failed=failed, skipped=skipped,
          )
          return ReindexResponse(
              reindexed=reindexed, failed=failed, skipped=skipped,
              by_type=ReindexByType(**by_type),
          )


      @router.delete(
          "/{entity_type}/{entity_id}",
          response_model=DeleteEmbeddingResponse,
          summary="Clear embedding for a specific entity (admin-only, experimental)",
      )
      async def delete_embedding(
          admin: CurrentAdmin,
          entity_type: str = Path(...),
          entity_id: str = Path(...),
      ) -> DeleteEmbeddingResponse:
          if entity_type not in ENTITY_TYPES:
              raise HTTPException(
                  status_code=400,
                  detail={
                      "type": "https://openhub.dev/problems/invalid-entity-type",
                      "title": "Invalid entity type",
                      "status": 400,
                      "detail": f"Unknown entity_type '{entity_type}'. Valid: {ENTITY_TYPES}",
                  },
              )
          svc = VectorSearchService(get_database())
          try:
              updated = svc.clear_embedding(entity_type, entity_id)
          except Exception as e:
              logger.error("delete_embedding_failed", entity_type=entity_type, id=entity_id, error=str(e)[:500])
              raise HTTPException(
                  status_code=503,
                  detail={
                      "type": "https://openhub.dev/problems/vector-store-unavailable",
                      "title": "Vector store unavailable",
                      "status": 503,
                      "detail": "Could not clear embedding; Turso may be unreachable.",
                  },
              )
          if not updated:
              raise HTTPException(
                  status_code=404,
                  detail={
                      "type": "https://openhub.dev/problems/entity-not-found",
                      "title": "Entity not found",
                      "status": 404,
                      "detail": f"No {entity_type} row with id '{entity_id}'.",
                  },
              )
          logger.info(
              "embedding_cleared",
              admin_id=admin.agent_id if hasattr(admin, "agent_id") else None,
              entity_type=entity_type, id=entity_id,
          )
          return DeleteEmbeddingResponse(entity_type=entity_type, id=entity_id, status="deleted")
      ```
